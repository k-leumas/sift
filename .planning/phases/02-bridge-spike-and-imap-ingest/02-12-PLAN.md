  lock.ts:
    export const INGEST_LOCK_SEED = 815309;   // seed of hashtextextended: a 64-bit key per mailbox in the bigint key space (collision with the fixed keys above is about 2^-64)
    export class IngestSessionBusyError extends Error {}   // 'IngestSession.run is already running on this session'
    export interface IngestSession { readonly mailboxId: string; run<T>(fn: (scope: Scope) => Promise<T>, options?: WithMailboxOptions): Promise<T> }   // not re-entrant
    export type IngestLockResult<T> = { acquired: true; value: T } | { acquired: false };
    export async function withIngestLock<T>(db: AppDb, mailboxId: string, fn: (session: IngestSession) => Promise<T>): Promise<IngestLockResult<T>>;
      // InvalidMailboxIdError before any query; `select pg_try_advisory_lock(hashtextextended($1, $2))` with $1 = mailbox id, $2 = INGEST_LOCK_SEED;
      // finally: `select pg_advisory_unlock(hashtextextended($1, $2))` then release the client (release(error) when the client errored).
  status.ts:
