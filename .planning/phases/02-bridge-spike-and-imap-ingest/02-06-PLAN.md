  type UniqueKey<T> = Exclude<keyof InferSelectModel<T> & string, 'mailboxId' | 'id' | 'createdAt' | 'updatedAt'>
  ScopedTableApi<T>.insertOrIgnore(
    rows: readonly Omit<InferInsertModel<T>, 'mailboxId' | 'id'>[],
    options: { target: readonly UniqueKey<T>[] },
  ): Promise<InferSelectModel<T>[]>
    // INSERT ... ON CONFLICT (mailbox_id, ...target) DO NOTHING RETURNING *: returns ONLY the rows it inserted, in no promised order.
    // Repeated keys inside `rows` are fine (PostgreSQL skips the later ones). Existing rows are not touched, so their updated_at stays.
  ScopedTableApi<T>.upsert(
    rows: readonly Omit<InferInsertModel<T>, 'mailboxId' | 'id'>[],
    options: { target: readonly UniqueKey<T>[]; update: readonly UniqueKey<T>[] },   // update is non-empty
  ): Promise<(InferSelectModel<T> & { inserted: boolean })[]>
    // INSERT ... ON CONFLICT (mailbox_id, ...target) DO UPDATE SET <each update column> = EXCLUDED.<column> RETURNING *, (xmax = 0) AS inserted.
    // Throws TypeError('upsert rows repeat a conflict key') BEFORE any SQL when two rows share the target values, because PostgreSQL rejects
    // such a statement ("ON CONFLICT DO UPDATE command cannot affect row a second time"). Results come in no promised order.
  Both: mailbox_id is prepended to the conflict target and filled into every row; `mailboxId`/`id` keys in rows, target or update throw TypeError;
  empty input runs no SQL and returns []. Callers match results to inputs by the target-column values, never by index.
  Match<T>: each value may also be a readonly array (inArray); an empty array matches nothing and runs no SQL.
  Scope.messageBody: ScopedTableApi<typeof messageBody> & { deleteExpired(at: Date): Promise<number> }
