  export function toUidSet(uids: readonly number[]): string;   // sorted, deduplicated, compressed ranges, e.g. [1,2,3,7,9,10] -> "1:3,7,9:10"; throws on an empty list
  export function createFolderSource(client: ImapFlow): FolderSource;
    examine(folder)            -> client.mailboxOpen(folder, { readOnly: true }); returns { uidValidity: Number(m.uidValidity), uidNext: m.uidNext, exists: m.exists }
    fetchDates(range)          -> client.fetchAll(range, { uid: true, internalDate: true }, { uid: true })
    fetchHeaders(uids)         -> client.fetchAll(toUidSet(uids), { uid: true, internalDate: true, size: true, envelope: true, bodyStructure: true, headers: [...HEADER_FIELDS] }, { uid: true })
    listUids(range)            -> client.search({ uid: range }, { uid: true }) ([] when the server returns false)
    searchSince(since)         -> client.search({ since }, { uid: true })
    downloadText(uid, part, n) -> client.download(String(uid), part, { uid: true, maxBytes: n }); read the stream into a string;
                                  truncated = (meta.expectedSize ?? the part's BODYSTRUCTURE size) > n (confirm in imapflow dist whether maxBytes counts encoded bytes)
  Every method except examine throws `FolderSource.<method>: <folder> is not the examined folder` (or `no folder examined yet`) if no folder is open, or the open folder is not the one examined.
From 02-04 (extended here): test-imap.ts gains `setFlags(user, folder, uid, flags: string[])` (doveadm flags add).
